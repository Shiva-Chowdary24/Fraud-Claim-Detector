@router.get("/customer/resolve-email/{customer_id}")
def resolve_email(customer_id: str):
    cid = (customer_id or "").strip()
    if not cid:
        raise HTTPException(status_code=400, detail="customer_id required")
    user = users.find_one({"customer_id": cid})
    if not user or not user.get("email"):
        raise HTTPException(status_code=404, detail="email not found for customer_id")
    return {"email": (user["email"] or "").strip().lower()}

@router.post("/query")
def ask_query(data: QueryRequest):
    try:
        cid = (str(data.customer_id) or "").strip()
        if not cid:
            raise HTTPException(status_code=400, detail="customer_id is required")

        user_profile = users.find_one({"customer_id": cid})
        print(f"DEBUG: Received customer_id: {cid} (Type: {type(cid)})")
        print(f"DEBUG: User Profile Found: {user_profile}")

        if not user_profile:
            raise HTTPException(
                status_code=404,
                detail="Profile lookup failed in database. Check customer_id exists and is a string."
            )

        # Normalize/whitelist values
        email = (user_profile.get("email") or "").strip().lower()
        if not email:
            raise HTTPException(status_code=422, detail="User email missing in profile")
        user_name = (user_profile.get("full_name") or "").strip() or "Unknown User"

        query_doc = {
            "customer_id": cid,
            "user_name": user_name,
            "email": email,
            "subject": data.subject,
            "query": data.query,
            "status": "Pending",
            "reply": None,
            "timestamp": now(),
            "resolved_at": None,
        }

        result = queries.insert_one(query_doc)

        # Notify admin (best-effort)
        try:
            notifications.insert_one({
                "recipient_id": "ADMIN",
                "message": f"New Support Ticket: {data.subject} from {user_name}",
                "type": "new_query",
                "link": "/admin/customer-queries",
                "timestamp": now(),
                "read": False
            })
        except Exception as ne:
            print("WARN: notification insert failed:", ne)

        return {"message": "Query submitted successfully", "id": str(result.inserted_id)}
    except HTTPException:
        raise
    except Exception:
        print("ERROR in /query:", traceback.format_exc())
        raise HTTPException(status_code=500, detail="Internal Server Error")

@router.post("/admin/reply/{query_id}")
def reply_to_query(query_id: str, data: ReplyRequest, role: Optional[str] = Header(None)):
    verify_admin(role)
    try:
        if not ObjectId.is_valid(query_id):
            raise HTTPException(status_code=400, detail="Invalid query_id")

        oid = ObjectId(query_id)

        update_result = queries.update_one(
            {"_id": oid},
            {
                "$set": {
                    "status": "Resolved",
                    "reply": data.reply,
                    "resolved_at": now(),
                }
            }
        )
        if update_result.matched_count == 0:
            raise HTTPException(status_code=404, detail="Query not found")

        original_query = queries.find_one({"_id": oid})
        if not original_query:
            # Updated but not retrievable (rare)
            return {"message": "Reply saved, but original query not found for notification"}

        # Notify customer (best-effort)
        try:
            cust_email = (original_query.get("email") or "").strip().lower()
            subj = original_query.get("subject", "your query")
            if cust_email:
                notifications.insert_one({
                    "recipient_id": cust_email,
                    "message": f"Admin replied to your query: '{subj}'",
                    "type": "support_reply",
                    "link": "/customer/support-history",
                    "timestamp": now(),
                    "read": False
                })
        except Exception as ne:
            print("WARN: customer notification insert failed:", ne)

        return {"message": "Reply sent and status updated to Resolved"}
    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

import os, hashlib, hmac, random

# Updated Model to include full_name
class AuthRequest(BaseModel):
    email: str
    password: str
    full_name: str = None  # Optional for login, required for signup

def hash_password(password):
    salt = os.urandom(16)
    hashed = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, 100000)
    return salt.hex(), hashed.hex()

def verify_password(password, salt, stored):
    salt = bytes.fromhex(salt)
    new_hash = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, 100000)
    return hmac.compare_digest(new_hash.hex(), stored)

@router.post("/customer/register")
def register(req: AuthRequest):
    if users.find_one({"email": req.email}):
        raise HTTPException(400, "User already exists")

    # Generate unique 6-digit Customer ID
    while True:
        cust_id = str(random.randint(100000, 999999))
        if not users.find_one({"customer_id": cust_id}):
            break

    salt, hashed = hash_password(req.password)

    users.insert_one({
        "customer_id": cust_id,
        "full_name": req.full_name, # Storing the name
        "email": req.email,
        "password": hashed,
        "salt": salt,
        "role": "customer",
        "created_at": datetime.utcnow().isoformat()
    })

    return {"message": "Registered", "customer_id": cust_id, "full_name": req.full_name}

@router.post("/customer/login")
def login(req: AuthRequest):
    user = users.find_one({"email": req.email, "role": "customer"})

    if not user or not verify_password(req.password, user["salt"], user["password"]):
        raise HTTPException(401, "Invalid credentials")

    # Returning both the ID and the Full Name
    return {
        "message": "Login success", 
        "email": req.email, 
        "customer_id": user.get("customer_id"),
        "full_name": user.get("full_name") 
    }
