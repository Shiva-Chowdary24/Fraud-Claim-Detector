@router.get("/customer/claim-history/{customer_id}")
def get_claim_history(customer_id: str):
    try:
        # 3. Fetch all logs for this customer that are NOT 'Pending'
        # This includes both 'Approved' and 'Declined'
        query = {
            "customer_id": customer_id,
            "status": {"$in": ["Approved", "Declined"]}
        }
        
        # Sort by most recent first
        history = list(db.Fraud_Logs.find(query).sort("timestamp", -1))
        
        for item in history:
            item["_id"] = str(item["_id"]) # Convert ObjectId for React
            
        return history
    except Exception as e:
        return []
