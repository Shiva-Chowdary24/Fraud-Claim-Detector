import { useEffect, useState } from "react";
import API from "../services/api";
import AdminLayout from "../components/AdminLayout";
import {
  CheckCircle,
  XCircle,
  User,
  Activity,
  Wallet,
  Clock,
  ShieldCheck
} from "lucide-react";
import { toast } from "react-toastify";

function PolicyRequests() {
  const [requests, setRequests] = useState([]);
  const [loading, setLoading] = useState(true);

  // Fetch pending requests
  const fetchRequests = async () => {
    try {
      const res = await API.get("/admin/policy-requests");
      setRequests(res.data);
    } catch {
      toast.error("Failed to load requests from server");
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchRequests();
  }, []);

  // Approve policy
  const handleApprove = async (id) => {
    try {
      const res = await API.post(`/admin/policy-approve/${id}`);
      toast.success(`Approved! Policy ID: ${res.data.policy_id}`);
      setRequests((prev) => prev.filter((r) => r._id !== id));
    } catch {
      toast.error("Could not complete approval");
    }
  };

  // Decline policy
  const handleDecline = async (id) => {
    try {
      await API.post(`/admin/policy-decline/${id}`);
      toast.warn("Application Declined");
      setRequests((prev) => prev.filter((r) => r._id !== id));
    } catch {
      toast.error("Action failed");
    }
  };

  return (
    <AdminLayout>
      <div className="max-w-6xl mx-auto">
        <header className="mb-10 text-left">
          <h2 className="text-3xl font-black text-white flex items-center gap-3">
            <ShieldCheck className="text-blue-500" size={32} />
            Policy Approval Requests
          </h2>
          <p className="text-slate-400 mt-2 text-sm">
            Underwrite and verify customer insurance applications.
          </p>
        </header>

        {loading ? (
          <div className="flex justify-center py-20">
            <div className="animate-spin rounded-full h-10 w-10 border-t-2 border-blue-500" />
          </div>
        ) : requests.length === 0 ? (
          <div className="bg-[#111e32]/40 border border-dashed border-slate-800 rounded-[2rem] p-20 text-center">
            <Clock className="mx-auto text-slate-700 mb-4" size={48} />
            <p className="text-slate-500 italic">
              No new applications to review.
            </p>
          </div>
        ) : (
          <div className="grid grid-cols-1 gap-6">
            {requests.map((req) => {
              const isDividend = req.policy_mode === "DIVIDEND";

              return (
                <div
                  key={req._id}
                  className={`bg-[#111e32]/80 border rounded-[2.5rem] shadow-2xl
                    ${isDividend ? "border-emerald-500/30" : "border-slate-800"}`}
                >
                  <div className="p-8 grid grid-cols-1 lg:grid-cols-12 gap-8">

                    {/* Applicant */}
                    <div className="lg:col-span-3 space-y-3">
                      <div className="text-[10px] uppercase font-bold text-slate-500 flex items-center gap-2">
                        <User size={14} className="text-blue-400" />
                        Applicant
                      </div>
                      <p className="text-white font-bold text-xl">
                        {req.full_name}
                      </p>
                      <p className="text-slate-500 text-xs truncate">
                        {req.email}
                      </p>
                    </div>

                    {/* Financial */}
                    <div className="lg:col-span-3 space-y-3">
                      <div className="text-[10px] uppercase font-bold text-slate-500 flex items-center gap-2">
                        <Wallet size={14} className="text-emerald-400" />
                        Income
                      </div>
                      <p className="text-white font-bold text-lg">
                        ₹{Number(req.annual_income).toLocaleString()}
                      </p>
                      <p className="text-slate-400 text-sm">
                        {req.occupation}
                      </p>
                    </div>

                    {/* Medical */}
                    <div className="lg:col-span-3 space-y-3">
                      <div className="text-[10px] uppercase font-bold text-slate-500 flex items-center gap-2">
                        <Activity size={14} className="text-red-400" />
                        Medical History
                      </div>
                      <p className="text-slate-300 text-sm italic bg-black/20 p-3 rounded-xl">
                        "{req.medical_history || "No disclosures"}"
                      </p>
                    </div>

                    {/* Actions */}
                    <div className="lg:col-span-3 flex flex-col justify-center gap-3 text-right">

                      <p
                        className={`font-black text-sm ${
                          isDividend ? "text-emerald-400" : "text-blue-400"
                        }`}
                      >
                        {req.plan_name}
                      </p>

                      {/* ✅ FIXED BADGE */}
                      <span
                        className={`inline-block px-2 py-0.5 rounded-md
                          text-[9px] font-black uppercase tracking-tighter border
                          ${
                            isDividend
                              ? "bg-emerald-500/10 text-emerald-400 border-emerald-500/20"
                              : "bg-blue-500/10 text-blue-400 border-blue-500/20"
                          }`}
                      >
                        {isDividend ? "DIVIDEND" : "NORMAL"}
                      </span>

                      <div className="flex gap-2 mt-2">
                        <button
                          onClick={() => handleApprove(req._id)}
                          className={`flex-1 py-3 rounded-xl text-xs font-bold text-white
                            ${
                              isDividend
                                ? "bg-emerald-600 hover:bg-emerald-500"
                                : "bg-blue-600 hover:bg-blue-500"
                            }`}
                        >
                          <CheckCircle size={16} /> Approve
                        </button>
                        <button
                          onClick={() => handleDecline(req._id)}
                          className="p-3 bg-red-600/10 hover:bg-red-600 text-red-500 hover:text-white rounded-xl"
                        >
                          <XCircle size={18} />
                        </button>
                      </div>
                    </div>
                  </div>

                  <div className="bg-slate-900/40 px-8 py-3 text-[10px] text-slate-500 font-bold uppercase flex justify-between">
                    <span>
                      <Clock size={12} className="inline mr-1" />
                      Tenure: {req.tenure} Years
                    </span>
                    <span>
                      Nominee:{" "}
                      <span className="text-slate-300 font-bold">
                        {req.nominee_name}
                      </span>{" "}
                      ({req.nominee_relation})
                    </span>
                  </div>
                </div>
              );
            })}
          </div>
        )}
      </div>
    </AdminLayout>
  );
}

export default PolicyRequests;
