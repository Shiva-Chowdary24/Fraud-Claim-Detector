def now():
    """Return UTC naive datetime (consistent everywhere)."""
    return datetime.utcnow()

def verify_admin(role: str):
    if role != "admin":
        raise HTTPException(status_code=403, detail="Unauthorized: Admin access required.")

def log_admin_action(action: str, details: str):
    audit_logs.insert_one({
        "action": action,
        "details": details,
        "timestamp": datetime.utcnow(),
    })

@router.post("/admin/policy-approve/{request_id}")
def approve(request_id: str, role: str = Header(None)):
    verify_admin(role)

    try:
        req = policy_requests.find_one({"_id": ObjectId(request_id)})
        if not req:
            raise HTTPException(404, "Application not found")

        policy_master = policies.find_one({"plan_name": req.get("plan_name")})
        if not policy_master:
            raise HTTPException(404, "Policy not found")

        plan_name = req["plan_name"]
        prefix = plan_name[:3].upper()
        policy_id = f"PL-{prefix}-{random.randint(1000, 9999)}"

        issued_policies.insert_one({
            "policy_id": policy_id,
            "customer_id": req["customer_id"],
            "email": req.get("email"),
            "plan_name": plan_name,
            "plan_type": policy_master["plan_type"],
            "premium_amount": policy_master["premium_amount"],
            "total_claim_amount": policy_master["total_claim_amount"],
            "tenure": policy_master["tenure"],
            "description": policy_master["description"],
            "benefits": policy_master["benefits"],
            "status": "Active",
            "approved_at": now()
        })

        policy_requests.delete_one({"_id": ObjectId(request_id)})

        # ✅ Notify customer
        notifications.insert_one({
            "recipient_id": str(req["customer_id"]),
            "message": f"✅ Policy '{plan_name}' has been issued successfully.",
            "link": "/customer/issued-policies",
            "type": "policy_approved",
            "timestamp": now(),
            "read": False
        })

        log_admin_action(
            action="POLICY_ISSUED",
            details=f"Issued policy {plan_name} to customer {req['customer_id']}"
        )

        return {"message": "Policy approved", "policy_id": policy_id}

    except Exception as e:
        raise HTTPException(500, str(e))


@router.post("/admin/policy-decline/{request_id}")
def decline_policy(request_id: str, role: str = Header(None)):
    verify_admin(role)

    try:
        req = policy_requests.find_one({"_id": ObjectId(request_id)})
        if not req:
            raise HTTPException(404, "Application not found")

        customer_id = req["customer_id"]

        notifications.insert_one({
            "recipient_id": str(customer_id),
            "message": (
                "❌ Your policy request was declined. "
                "Please contact support for details."
            ),
            "link": "/customer/ask-question",
            "type": "policy_declined",
            "timestamp": now(),
            "read": False
        })

        log_admin_action(
            action="POLICY_DECLINED",
            details=f"Declined policy request for customer {customer_id}"
        )

        policy_requests.delete_one({"_id": ObjectId(request_id)})

        return {"message": "Policy request declined and customer notified"}

    except Exception as e:
        raise HTTPException(500, str(e))

@router.get("/admin/logs")
def get_fraud_logs(role: str = Header(None)):
    verify_admin(role)

    logs = list(fraud_logs.find({}))
    for log in logs:
        log["_id"] = str(log["_id"])

    def normalize_timestamp(ts):
        """
        Convert all timestamps to UTC naive to avoid:
        TypeError: can't compare offset-naive and offset-aware datetimes
        """
        if isinstance(ts, str):
            try:
                dt = datetime.fromisoformat(ts)
            except Exception:
                return now()
        elif isinstance(ts, datetime):
            dt = ts
        else:
            return now()

        if dt.tzinfo:
            dt = dt.astimezone(timezone.utc).replace(tzinfo=None)

        return dt

    return sorted(
        logs,
        key=lambda x: (
            x.get("verified", False),
            normalize_timestamp(x.get("timestamp"))
        ),
        reverse=True
    )
