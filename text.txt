@router.get("/admin/queries")
def get_queries(role: str = Header(None)):
    verify_admin(role)
    results = list(queries.find())
    for r in results:
        r["_id"] = str(r["_id"])
    return results

@router.post("/query")
def ask_query(data: dict):
    data["status"] = "Pending"
    data["timestamp"] = now()
    queries.insert_one(data)
    return {"message": "Query sent"}
