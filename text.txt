import React, { useEffect, useState } from "react";
import CustSidebar from "../components/CustSidebar";
import CustNavbar from "../components/CustNavbar";
import API from "../services/api";
import {
  MessageSquare,
  Clock,
  CheckCircle2,
  HelpCircle,
  Loader2,
  AlertCircle,
  RefreshCw,
  Info,
} from "lucide-react";
import { toast } from "react-toastify";

function CustSupportHistory() {
  const [history, setHistory] = useState([]);
  const [loading, setLoading] = useState(true);
  const [debug, setDebug] = useState({
    email: "",
    encodedEmail: "",
    url: "",
    lastFetchedAt: "",
  });

  const fetchHistory = async () => {
    setLoading(true);
    try {
      // 1) Try to read email from localStorage
      let rawEmail = localStorage.getItem("user_email");
      let email = (rawEmail || "").trim().toLowerCase();

      // 2) Fallback: resolve email by customer_id from backend if missing
      if (!email) {
        const rawCid = localStorage.getItem("customer_id");
        const customerId = (rawCid || "").trim();
        if (customerId) {
          try {
            const resolveRes = await API.get(
              `/customer/resolve-email/${encodeURIComponent(customerId)}`
            );
            email = (resolveRes?.data?.email || "").trim().toLowerCase();
            if (email) {
              localStorage.setItem("user_email", email); // cache for next time
            }
          } catch (e) {
            // silently continue; we'll show a toast below if we still don't have email
          }
        }
      }

      if (!email) {
        toast.error("User email not found. Please log in again.");
        setHistory([]);
        setDebug((d) => ({
          ...d,
          email: "",
          encodedEmail: "",
          url: "",
          lastFetchedAt: new Date().toLocaleString(),
        }));
        return;
      }

      // 3) Fetch history using normalized + encoded email
      const encodedEmail = encodeURIComponent(email);
      const path = `/customer/queries/${encodedEmail}`;
      const url = API?.defaults?.baseURL ? API.defaults.baseURL + path : path;

      const res = await API.get(path);
      const data = Array.isArray(res.data) ? res.data : [];

      setHistory(data);
      setDebug({
        email,
        encodedEmail,
        url,
        lastFetchedAt: new Date().toLocaleString(),
      });
    } catch (err) {
      console.error("Fetch error:", err);
      toast.error("Could not load support history.");
      setHistory([]);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchHistory();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const cardStyle =
    "bg-[#111e32]/60 backdrop-blur-md border border-white/5 rounded-[2rem] p-8 shadow-xl mb-6 transition-all hover:border-blue-500/20";

  return (
    <div className="flex min-h-screen bg-[#0a1628] text-slate-200 font-sans">
      <CustSidebar />
      <div className="flex-1 flex flex-col">
        <CustNavbar />
        <main className="p-10 max-w-5xl mx-auto w-full text-left">
          {/* Header + Manual Refresh */}
          <header className="mb-8">
            <div className="flex items-center justify-between">
              <h1 className="text-4xl font-black text-white tracking-tighter italic uppercase flex items-center gap-3">
                <Clock className="text-blue-500" size={32} /> Support History
              </h1>
              <button
                onClick={fetchHistory}
                className="inline-flex items-center gap-2 bg-blue-600 hover:bg-blue-500 text-white px-4 py-2 rounded-xl text-xs font-black uppercase tracking-widest"
              >
                <RefreshCw size={14} />
                Refresh
              </button>
            </div>
            <p className="text-slate-500 text-[10px] tracking-[0.2em] mt-2 uppercase font-bold">
              Tracking your communication with our underwriting team
            </p>
          </header>

          {/* Debug Strip */}
          <div className="mb-6 bg-black/30 border border-white/5 rounded-2xl p-4 text-xs">
            <div className="flex items-center gap-2 mb-2">
              <Info className="text-slate-400" size={14} />
              <span className="text-slate-400 font-bold uppercase tracking-widest">
                Debug
              </span>
            </div>
            <div className="grid grid-cols-1 md:grid-cols-2 gap-2 text-slate-400">
              <div>
                <span className="text-slate-500">Email: </span>
                <code className="text-slate-300">{debug.email || "-"}</code>
              </div>
              <div>
                <span className="text-slate-500">Encoded: </span>
                <code className="text-slate-300">{debug.encodedEmail || "-"}</code>
              </div>
              <div className="md:col-span-2">
                <span className="text-slate-500">GET: </span>
                <code className="text-slate-300 break-all">{debug.url || "-"}</code>
              </div>
              <div className="md:col-span-2">
                <span className="text-slate-500">Records: </span>
                <code className="text-slate-300">{history.length}</code>
                <span className="text-slate-500 ml-4">Last Fetched: </span>
                <code className="text-slate-300">{debug.lastFetchedAt}</code>
              </div>
            </div>
          </div>

          {loading ? (
            <div className="flex flex-col items-center justify-center py-20">
              <Loader2 className="animate-spin text-blue-500 mb-4" size={40} />
              <p className="text-slate-500 text-xs uppercase tracking-widest">
                Accessing Archives...
              </p>
            </div>
          ) : history && history.length > 0 ? (
            <div className="space-y-4">
              {history.map((item) => (
                <div key={item._id} className={cardStyle}>
                  {/* Status & Date Header */}
                  <div className="flex justify-between items-start mb-6">
                    <div>
                      <span
                        className={`px-4 py-1 rounded-full text-[9px] font-black uppercase tracking-[0.15em] border ${
                          item.status === "Resolved"
                            ? "bg-emerald-500/10 text-emerald-400 border-emerald-500/20"
                            : "bg-amber-500/10 text-amber-400 border-amber-500/20"
                        }`}
                      >
                        {item.status}
                      </span>
                      <h3 className="text-xl font-bold text-white mt-4 tracking-tight">
                        {item.subject || "General Inquiry"}
                      </h3>
                    </div>
                    <div className="text-right">
                      <p className="text-slate-600 text-[9px] font-black uppercase tracking-tighter">
                        Submitted On
                      </p>
                      <p className="text-slate-400 text-xs font-bold">
                        {item.timestamp
                          ? new Date(item.timestamp).toLocaleDateString()
                          : "-"}
                      </p>
                    </div>
                  </div>

                  {/* Original Query */}
                  <div className="bg-black/20 p-5 rounded-2xl border border-white/5 relative">
                    <MessageSquare
                      size={14}
                      className="absolute -top-2 -left-2 text-slate-700 bg-[#0a1628] rounded-full"
                    />
                    <p className="text-slate-400 text-sm leading-relaxed italic">
                      {`"${item.query}"`}
                    </p>
                  </div>

                  {/* Admin Reply Section */}
                  {typeof item.reply === "string" && item.reply.trim().length > 0 ? (
                    <div className="mt-8 pt-8 border-t border-white/5">
                      <div className="flex items-center gap-2 mb-4">
                        <div className="bg-blue-500/20 p-1.5 rounded-lg">
                          <CheckCircle2 size={14} className="text-blue-400" />
                        </div>
                        <span className="text-[10px] font-black text-blue-400 uppercase tracking-widest">
                          Resolution Details
                        </span>
                      </div>
                      <div className="bg-blue-500/5 p-6 rounded-2xl border border-blue-500/10">
                        <p className="text-white text-sm leading-relaxed font-medium">
                          {item.reply}
                        </p>
                      </div>
                      <div className="mt-4 flex justify-end">
                        <p className="text-slate-600 text-[9px] uppercase font-bold tracking-tighter italic">
                          Closed:{" "}
                          {new Date(item.resolved_at || item.timestamp).toLocaleString()}
                        </p>
                      </div>
                    </div>
                  ) : (
                    <div className="mt-6 flex items-center gap-2 text-slate-600">
                      <AlertCircle size={14} />
                      <p className="text-[10px] font-bold uppercase tracking-widest italic">
                        Awaiting response from Underwriter
                      </p>
                    </div>
                  )}
                </div>
              ))}
            </div>
          ) : (
            <>
              <div className="bg-[#111e32]/40 border border-dashed border-slate-800 rounded-[3rem] p-20 text-center">
                <HelpCircle className="mx-auto text-slate-800 mb-4" size={50} />
                <p className="text-slate-600 font-bold uppercase tracking-widest text-sm">
                  No support tickets found in your account.
                </p>
              </div>

              {/* Raw data preview for quick sanity */}
              <div className="mt-6 text-xs text-slate-400">
                <p className="mb-2 font-bold uppercase tracking-widest flex items-center gap-2">
                  <Info size={14} /> Debug: Raw Data Preview
                </p>
                <pre className="bg-black/30 border border-white/5 rounded-xl p-4 overflow-auto">
                  {JSON.stringify(history, null, 2)}
                </pre>
              </div>
            </>
          )}
        </main>
      </div>
    </div>
  );
}

export default CustSupportHistory;s
