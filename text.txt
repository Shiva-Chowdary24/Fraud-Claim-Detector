class StatusUpdate(BaseModel):
    Policy_id: str
    status: str
    reason: Optional[str] = ""


@router.post("/admin/logs/update-status")
def update_log_status(data: StatusUpdate, role: str = Header(None)):
    verify_admin(role)

    try:
        log_entry = fraud_logs.find_one({"policy_id": data.Policy_id})
        if not log_entry:
            raise HTTPException(404, "Claim log not found")

        cust_id = log_entry["customer_id"]

        fraud_logs.update_one(
            {"_id": log_entry["_id"]},
            {
                "$set": {
                    "status": data.status,
                    "verified": True,
                    "admin_reason": data.reason,
                    "updated_at": now()
                }
            }
        )

        # ✅ Notify customer ONLY (correct behavior)
        link = (
            "/customer/amount-predict"
            if data.status.lower() == "approved"
            else "/customer/ask-question"
        )

        msg = (
            f"✅ Your claim for Policy {data.Policy_id} has been approved."
            if data.status.lower() == "approved"
            else f"❌ Your claim for Policy {data.Policy_id} has been {data.status}."
        )

        notifications.insert_one({
            "recipient_id": str(cust_id),
            "message": msg,
            "link": link,
            "type": "claim_update",
            "timestamp": now(),
            "read": False
        })

        log_admin_action(
            action="CLAIM_STATUS_UPDATED",
            details=f"Policy {data.Policy_id} marked {data.status}"
        )

        return {"message": f"Claim {data.status} successfully"}

    except Exception as e:
        raise HTTPException(500, str(e))
