from fastapi import APIRouter, HTTPException, Header

from services.admin_service import AdminService

from pydantic import BaseModel

from typing import Optional

router = APIRouter(tags=["Admin Operations"])

def check_auth(role: str):

    if not AdminService.verify_admin_role(role):

        raise HTTPException(status_code=403, detail="Unauthorized: Admin access required.")

@router.get("/admin/policy-requests")

def get_requests(role: str = Header(None)):

    check_auth(role)

    try:

        return AdminService.get_all_pending_requests()

    except Exception as e:

        raise HTTPException(500, f"Error: {str(e)}")

@router.post("/admin/policy-approve/{request_id}")

def approve(request_id: str, role: str = Header(None)):

    check_auth(role)

    policy_id, error = AdminService.approve_policy_request(request_id)

    if error:

        status_code = 404 if "not found" in error.lower() else 500

        raise HTTPException(status_code=status_code, detail=error)

    return {

        "message": "Approved",

        "policy_id": policy_id

    }

@router.get("/admin/dashboard")

def get_stats(role: str = Header(None)):

    check_auth(role)

    try:

        return AdminService.get_dashboard_stats()

    except Exception as e:

        raise HTTPException(500, detail="Failed to fetch stats")

@router.get("/admin/audit-logs")

def get_audit_logs(role: str = Header(None)):

    from database import audit_logs # Local import to avoid circularity

    check_auth(role)

    return list(audit_logs.find({}, {"_id": 0}).sort("timestamp", -1))
