@app.post("/notifications/add")
def add_notification(data: NotificationIn):
    notification_doc = {
        "recipient_id": data.recipient_id,
        "message": data.message,
        "link": data.link,
        "status": data.status,
        "read": False, # Initial state
        "timestamp": datetime.utcnow().isoformat()
    }
    db.notifications.insert_one(notification_doc)
    return {"message": "Notification dispatched"}
