@router.post("/query")
def ask_query(data: QueryRequest):
    print("🔥 DEBUG: /query endpoint HIT")

    try:
        print("🔍 DEBUG: Request body:", data)

        cid = data.customer_id.strip()
        print("✅ DEBUG: customer_id:", cid)

        # --- USER LOOKUP (OPTIONAL) ---
        user_profile = users.find_one({
            "$or": [
                {"customer_id": cid},
                {"customer_id": int(cid)} if cid.isdigit() else {}
            ]
        })

        print("🔍 DEBUG: user_profile:", user_profile)

        if user_profile:
            user_name = (user_profile.get("full_name") or "").strip() or "Unknown User"
            email = (user_profile.get("email") or "").strip().lower()
        else:
            print("⚠️ DEBUG: No user profile found — using fallback")
            user_name = "Guest User"
            email = "unknown@support.local"

        query_doc = {
            "customer_id": cid,
            "user_name": user_name,
            "email": email,
            "subject": data.subject,
            "query": data.query,
            "status": "Pending",
            "reply": None,
            "timestamp": datetime.utcnow(),
            "resolved_at": None,
        }

        print("📝 DEBUG: Inserting query:", query_doc)

        result = queries.insert_one(query_doc)

        print("✅ DEBUG: Query inserted:", result.inserted_id)

        # --- ADMIN NOTIFICATION ---
        try:
            notifications.insert_one({
                "recipient_id": "ADMIN",
                "message": f"New Support Ticket: {data.subject} from {user_name}",
                "type": "new_query",
                "link": "/admin/customer-queries",
                "timestamp": datetime.utcnow(),
                "read": False
            })
            print("✅ DEBUG: Admin notified")
        except Exception as ne:
            print("⚠️ WARN: Notification failed:", ne)

        return {
            "message": "Query submitted successfully",
            "id": str(result.inserted_id)
        }

    except HTTPException:
        raise
    except Exception:
        print("🔥 ERROR in /query:")
        print(traceback.format_exc())
        raise HTTPException(status_code=500, detail="Internal Server Error")
